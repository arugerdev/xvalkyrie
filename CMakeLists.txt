cmake_minimum_required(VERSION 3.25)
project(XValkyrie VERSION 0.0.9 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "XValkyrie builds for 64-bit Windows with MSVC")
endif()
include(CTest)
include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)
option(XV_VULKAN "Build the headless Vulkan backend and pixel check (requires Vulkan SDK/loader)" OFF)
option(XV_DXVK "Build Direct3D 11 on Vulkan for the vulkan backend (requires meson and glslang)" OFF)
add_library(xv_options INTERFACE)
target_include_directories(xv_options INTERFACE src)
target_compile_options(xv_options INTERFACE /W4 /permissive-)
set(xv_version_header "${PROJECT_BINARY_DIR}/generated/app/version.hpp")
add_custom_target(xv_version_header
  COMMAND ${CMAKE_COMMAND} "-DSOURCE_DIR=${PROJECT_SOURCE_DIR}" "-DOUTPUT=${xv_version_header}"
    -DVERSION=${PROJECT_VERSION} -P "${PROJECT_SOURCE_DIR}/cmake/Version.cmake"
  BYPRODUCTS "${xv_version_header}" VERBATIM)
add_library(xv_version INTERFACE)
target_include_directories(xv_version INTERFACE "${PROJECT_BINARY_DIR}/generated")
add_dependencies(xv_version xv_version_header)
if(XV_DXVK)
  include(cmake/Dxvk.cmake)
endif()
add_subdirectory(src)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
